Extension Distribution & Onboarding Guide

Provide authorized extension access to your students:

  1. Open Google Chrome and navigate to chrome://extensions/.
  2. Toggle "Developer mode" ON in the top-right corner.
  3. Click "Load unpacked" and choose the extension build folder.
  4. Pin "LMS Java Coding Helper" to the browser toolbar.
  5. When prompted, students enter their User ID and Password from this portal.
  6. Single-device binding locks automatically upon their first login!

1. Cloud Firestore Rules (Recommended)

Deploy in Firebase Console > Firestore Rules ↗ 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /admins/{adminId} {
      allow list: if false;
      allow get: if true;
      allow create, update: if true;
      allow delete: if false;
    }
    match /users/{userId} {
      allow get, list: if true;
      allow create, update, delete: if true;
    }
    match /system_settings/{settingId} {
      allow read, write: if true;
    }
    match /activity_logs/{logId} {
      allow read, create: if true;
      allow update, delete: if false;
    }
  }
}

2. Realtime Database Rules (Locked Root Listing)

Deploy in Firebase Console > RTDB Rules ↗ 

{
  "rules": {
    "admins": {
      ".read": false,
      "$adminId": {
        ".read": true,
        ".write": true
      }
    },
    "users": {
      ".read": true,
      ".write": true
    },
    "system_settings": {
      ".read": true,
      ".write": true
    },
    "activity_logs": {
      ".read": true,
      "$logId": {
        ".write": "!data.exists()"
      }
    }
  }
}
